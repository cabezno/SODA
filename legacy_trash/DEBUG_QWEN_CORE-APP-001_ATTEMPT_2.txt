#include <windows.h>
#include <iostream>

class MyWindowsUIFramework {
public:
    virtual ~MyWindowsUIFramework() = default;

protected:
    virtual LRESULT WndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
        switch (msg) {
            case WM_DESTROY:
                PostQuitMessage(0);
                return 0;
            default:
                return DefWindowProc(hwnd, msg, wParam, lParam);
        }
    }

public:
    void Initialize() {
        WNDCLASSEX wcex = { sizeof(WNDCLASSEX) };
        wcex.lpfnWndProc = &MyWindowsUIFramework::StaticWndProc; // Use static member function for lpfnWndProc
        wcex.hInstance = GetModuleHandle(nullptr);
        wcex.lpszClassName = TEXT("MyAppClass");
        wcex.cbWndExtra = sizeof(LONG_PTR); // Allocate space for the window instance pointer

        if (!RegisterClassEx(&wcex)) {
            std::cerr << "Failed to register window class." << std::endl;
            return;
        }

        HWND hwnd = CreateWindow(TEXT("MyAppClass"), TEXT("My Window"),
                                 WS_OVERLAPPEDWINDOW,
                                 CW_USEDEFAULT, 0, CW_USEDEFAULT, 0, nullptr, nullptr,
                                 wcex.hInstance, this); // Pass 'this' as lpParam

        if (!hwnd) {
            std::cerr << "Failed to create window." << std::endl;
            return;
        }

        SetWindowLongPtr(hwnd, GWLP_USERDATA, (LONG_PTR)this); // Store the instance pointer in window's userdata

        ShowWindow(hwnd, SW_SHOW);
        UpdateWindow(hwnd);

        MSG msg = {};
        while (GetMessage(&msg, nullptr, 0, 0)) {
            TranslateMessage(&msg);
            DispatchMessage(&msg);
        }
    }

    static LRESULT CALLBACK StaticWndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
        MyWindowsUIFramework* instance = reinterpret_cast<MyWindowsUIFramework*>(GetWindowLongPtr(hwnd, GWLP_USERDATA));
        if (instance) {
            return instance->WndProc(hwnd, msg, wParam, lParam);
        }
        return DefWindowProc(hwnd, msg, wParam, lParam);
    }

private:
    // Other members and methods
};

// Example usage
int main() {
    MyWindowsUIFramework framework;
    framework.Initialize();
    return 0;
}
