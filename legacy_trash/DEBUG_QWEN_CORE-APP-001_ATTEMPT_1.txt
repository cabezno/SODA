#include <windows.h>

// Define the abstract type for a Windows UI Framework
class WindowsUIFramework {
public:
    virtual ~WindowsUIFramework() {}
    virtual bool Initialize(HINSTANCE hInstance, int nCmdShow) = 0;
    virtual void RunMessageLoop() = 0;
    virtual LRESULT CALLBACK WndProc(HWND hwnd, UINT uMsg, WPARAM wParam, LPARAM lParam) = 0;
};

// Concrete implementation of the Windows UI Framework
class MyWindowsUIFramework : public WindowsUIFramework {
public:
    bool Initialize(HINSTANCE hInstance, int nCmdShow) override {
        // Create the main window
        WNDCLASSEX wcex;
        wcex.cbSize = sizeof(WNDCLASSEX);
        wcex.style          = CS_HREDRAW | CS_VREDRAW;
        wcex.lpfnWndProc    = &MyWindowsUIFramework::WindowProc;
        wcex.cbClsExtra     = 0;
        wcex.cbWndExtra     = 0;
        wcex.hInstance      = hInstance;
        wcex.hIcon          = LoadIcon(hInstance, IDI_APPLICATION);
        wcex.hCursor        = LoadCursor(nullptr, IDC_ARROW);
        wcex.hbrBackground  = (HBRUSH)(COLOR_WINDOW+1);
        wcex.lpszMenuName   = nullptr;
        wcex.lpszClassName  = L"WindowsDesktopApp";
        wcex.hIconSm        = LoadIcon(wcex.hInstance, IDI_APPLICATION);

        if (!RegisterClassEx(&wcex)) {
            return false;
        }

        m_hwnd = CreateWindow(L"WindowsDesktopApp", L"My Windows Desktop App",
                              WS_OVERLAPPEDWINDOW,
                              CW_USEDEFAULT, 0,
                              CW_USEDEFAULT, 0,
                              nullptr, nullptr, hInstance, this);

        if (!m_hwnd) {
            return false;
        }

        ShowWindow(m_hwnd, nCmdShow);
        UpdateWindow(m_hwnd);

        return true;
    }

    void RunMessageLoop() override {
        MSG msg;
        while (GetMessage(&msg, nullptr, 0, 0)) {
            TranslateMessage(&msg);
            DispatchMessage(&msg);
        }
    }

    LRESULT CALLBACK WndProc(HWND hwnd, UINT uMsg, WPARAM wParam, LPARAM lParam) override {
        switch (uMsg) {
            case WM_DESTROY:
                PostQuitMessage(0);
                break;
            default:
                return DefWindowProc(hwnd, uMsg, wParam, lParam);
        }
        return 0;
    }

private:
    HWND m_hwnd = nullptr;
};

// Function to create and run the desktop UI framework
WindowsUIFramework* CreateDesktopUIFramework() {
    return new MyWindowsUIFramework();
}
