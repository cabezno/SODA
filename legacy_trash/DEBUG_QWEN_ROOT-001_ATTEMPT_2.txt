#include <windows.h>
#include <queue>
#include <string>

class CountdownTimerUI {
public:
    void Initialize(HINSTANCE hInstance);
    void UpdateDisplay(const std::string& time, const std::string& state);
    void AddEvent(const std::string& event);

private:
    HWND hwnd;
    std::queue<std::string> eventsQueue;

    static LRESULT CALLBACK WndProc(HWND, UINT, WPARAM, LPARAM);

    void CreateWindow();
    void UpdateTimeDisplay(const std::string& time);
    void UpdateButtonStates(const std::string& state);
};

void CountdownTimerUI::Initialize(HINSTANCE hInstance) {
    CreateWindow();
}

void CountdownTimerUI::UpdateDisplay(const std::string& time, const std::string& state) {
    UpdateTimeDisplay(time);
    UpdateButtonStates(state);
}

void CountdownTimerUI::AddEvent(const std::string& event) {
    eventsQueue.push(event);
}

LRESULT CALLBACK CountdownTimerUI::WndProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
    switch (msg) {
        case WM_COMMAND:
            // Handle button clicks
            if (LOWORD(wParam) == ID_PLAY || LOWORD(wParam) == ID_PAUSE || LOWORD(wParam) == ID_STOP || LOWORD(wParam) == ID_RESET) {
                std::string event = LOWORD(wParam) == ID_SET_DURATION ? "SET_DURATION:" + std::to_string(HIWORD(wParam)) : GetEventString(LOWORD(wParam));
                AddEvent(event);
            }
            break;
        case WM_DESTROY:
            PostQuitMessage(0);
            break;
        default:
            return DefWindowProc(hwnd, msg, wParam, lParam);
    }
    return 0;
}

void CountdownTimerUI::CreateWindow() {
    WNDCLASSEX wcex = { sizeof(WNDCLASSEX), CS_HREDRAW | CS_VREDRAW, WndProc, 0, 0, GetModuleHandle(nullptr), LoadIcon(nullptr, IDI_APPLICATION), LoadCursor(nullptr, IDC_ARROW), (HBRUSH)(COLOR_WINDOW + 1), nullptr, "CountdownTimer", nullptr };
    RegisterClassEx(&wcex);
    hwnd = CreateWindow("CountdownTimer", "Countdown Timer", WS_OVERLAPPEDWINDOW, CW_USEDEFAULT, 0, 320, 240, nullptr, nullptr, GetModuleHandle(nullptr), this);
    ShowWindow(hwnd, SW_SHOW);
    UpdateWindow(hwnd);
}
