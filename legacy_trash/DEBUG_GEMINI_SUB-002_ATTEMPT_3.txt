```python
import json

def implement_html_structure(page_layout_plan: dict, product_info: dict) -> str:
    title = page_layout_plan.get("title_template", "{product_name} - Landing Page").format(
        product_name=product_info.get("product_name", "Our Product")
    )

    header_content_parts = []
    main_content_parts = []
    footer_content_parts = []

    for section_plan in page_layout_plan.get("sections", []):
        section_type = section_plan.get("type")
        content_areas = section_plan.get("content_areas", [])

        if section_type == "header":
            header_content_parts.append('    <header>')
            if "logo" in content_areas:
                header_content_parts.append(f'        <div class="logo"><h1>{product_info.get("product_name", "Our Product")}</h1></div>')
            if "nav" in content_areas:
                header_content_parts.append('        <nav>')
                header_content_parts.append('            <ul>')
                header_content_parts.append('                <li><a href="#features">Features</a></li>')
                header_content_parts.append('                <li><a href="#about">About</a></li>')
                header_content_parts.append('                <li><a href="#contact">Contact</a></li>')
                header_content_parts.append('            </ul>')
                header_content_parts.append('        </nav>')
            header_content_parts.append('    </header>')
        
        elif section_type == "hero":
            main_content_parts.append('    <section class="hero">')
            if "headline" in content_areas:
                main_content_parts.append(f'        <h1>{product_info.get("headline", "Welcome to Our Product")}</h1>')
            if "tagline" in content_areas:
                main_content_parts.append(f'        <p>{product_info.get("tagline", "The best solution for your needs.")}</p>')
            if "cta_button" in content_areas:
                main_content_parts.append(f'        <a href="#" class="button">{product_info.get("cta_text", "Learn More")}</a>')
            main_content_parts.append('    </section>')

        elif section_type == "features":
            main_content_parts.append('    <section id="features" class="features">')
            main_content_parts.append('        <h2>Key Features</h2>')
            main_content_parts.append('        <div class="feature-list">')
            for feature in product_info.get("features", []):
                main_content_parts.append('            <div class="feature-item">')
                main_content_parts.append(f'                <h3>{feature.get("title", "Feature Title")}</h3>')
                main_content_parts.append(f'                <p>{feature.get("description", "Feature description goes here.")}</p>')
                main_content_parts.append('            </div>')
            main_content_parts.append('        </div>')
            main_content_parts.append('    </section>')

        elif section_type == "about":
            main_content_parts.append('    <section id="about" class="about">')
            main_content_parts.append('        <h2>About Us</h2>')
            main_content_parts.append(f'        <p>{product_info.get("about_text", "Lorem ipsum dolor sit amet, consectetur adipiscing elit.")}</p>')
            main_content_parts.append('    </section>')

        elif section_type == "call_to_action":
            main_content_parts.append('    <section class="call-to-action">')
            if "cta_headline" in content_areas:
                main_content_parts.append(f'        <h2>{product_info.get("cta_headline", "Ready to Get Started?")}</h2>')
            if "cta_button" in content_areas:
                main_content_parts.append(f'        <a href="#" class="button primary">{product_info.get("cta_text", "Sign Up Now")}</a>')
            main_content_parts.append('    </section>')

        elif section_type == "footer":
            footer_content_parts.append('    <footer>')
            if "copyright" in content_areas:
                year = product_info.get("copyright_year", "YYYY")
                company = product_info.get("company_name", "Your Company")
                footer_content_parts.append(f'        <p>&copy; {year} {company}. All rights reserved.</p>')
            if "links" in content_areas:
                footer_content_parts.append('        <nav>')
                footer_content_parts.append('            <ul>')
                footer_content_parts.append('                <li><a href="#">Privacy Policy</a></li>')
                footer_content_parts.append('                <li><a href="#">Terms of Service</a></li>')
                footer_content_parts.append('            </ul>')
                footer_content_parts.append('        </nav>')
            footer_content_parts.append('    </footer>')

    final_html_parts = [
        "<!DOCTYPE html>",
        '<html lang="en">',
        '<head>',
        '    <meta charset="UTF-8">',
        '    <meta name="viewport" content="width=device-width, initial-scale=1.0">',
        f'    <title>{title}</title>',
        '</head>',
        '<body>'
    ]

    final_html_parts.extend(header_content_parts)
    
    if main_content_parts:
        final_html_parts.append('    <main>')
        final_html_parts.extend(main_content_parts)
        final_html_parts.append('    </main>')

    final_html_parts.extend(footer_content_parts)

    final_html_parts.append('</body>')
    final_html_parts.append('</html>')

    return "\n".join(final_html_parts)
```
