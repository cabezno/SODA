#include <vector>
#include <string>

class DataSource {
public:
    enum class Type { WEB_LINK, LOCAL_FILE };
    std::string url_or_path;
    Type type;

    DataSource(std::string url_or_path, Type type) : url_or_path(url_or_path), type(type) {}
};

class RAGDataSourceManager {
public:
    void addUserConfiguredRAGSource(const std::string& url_or_path, DataSource::Type type);
    std::vector<DataSource> getUserConfiguredRAGSources() const;

private:
    std::vector<DataSource> rag_sources;
};

void RAGDataSourceManager::addUserConfiguredRAGSource(const std::string& url_or_path, DataSource::Type type) {
    rag_sources.emplace_back(url_or_path, type);
}

std::vector<DataSource> RAGDataSourceManager::getUserConfiguredRAGSources() const {
    return rag_sources;
}
