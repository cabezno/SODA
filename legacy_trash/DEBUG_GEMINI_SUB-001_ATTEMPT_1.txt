    <div id="root"
         data-product-info='{
             "title": "Awesome Product",
             "headline": "Unlock Your Potential with Our Revolutionary Product!",
             "description": "Experience unparalleled efficiency and innovation. Our product is designed to streamline your workflow and elevate your results.",
             "imageUrl": "https://via.placeholder.com/800x450/007bff/ffffff?text=Product+Showcase",
             "imageAlt": "Product Showcase Image",
             "features": [
                 "Seamless Integration with Existing Tools",
                 "Intuitive User Interface for Easy Adoption",
                 "Robust Security Features to Protect Your Data",
                 "24/7 Customer Support for Uninterrupted Service",
                 "Scalable Solutions for Growing Businesses"
             ]
         }'
         data-api-endpoint="https://api.example.com/lead-capture">
    </div>

    <!-- React and Babel for JSX/TSX Transpilation -->
    <script src="https://unpkg.com/react/umd/react.development.js"></script>
    <script src="https://unpkg.com/react-dom/umd/react-dom.development.js"></script>
    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

    <script type="text/babel" data-presets="es2015,react,typescript">
        // IMPORTANT: This script block uses 'text/babel' and 'data-presets="typescript"'
        // to simulate TypeScript compilation directly in the browser.
        // For production, a proper build setup (e.g., Webpack/Vite with ts-loader)
        // would compile TypeScript to JavaScript beforehand.

        const { useState } = React;

        // Define interfaces for props and content data
        interface ProductInfoContent {
            title: string;
            headline: string;
            description: string;
            imageUrl: string;
            imageAlt: string;
            features: string[];
        }

        interface LandingPageProps {
            product_info_content: ProductInfoContent;
            backend_api_endpoint: string;
        }

        const LandingPage: React.FC<LandingPageProps> = ({ product_info_content, backend_api_endpoint }) => {
            const [email, setEmail] = useState<string>('');
            const [message, setMessage] = useState<string>('');
            const [isSubmitting, setIsSubmitting] = useState<boolean>(false);

            const handleSubmit = async (event: React.FormEvent) => {
                event.preventDefault();
                setMessage('');
                setIsSubmitting(true);

                try {
                    // Simulate API call
                    // In a real application, this would send data to backend_api_endpoint
                    const response = await fetch(backend_api_endpoint, {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                        },
                        body: JSON.stringify({ email }),
                    });

                    if (response.ok) {
                        setMessage('Thank you for subscribing!');
                        setEmail('');
                    } else {
                        // Attempt to parse error message from response
                        const errorData = await response.json().catch(() => ({ message: 'Server error' }));
                        setMessage(`Submission failed: ${errorData.message || 'Unknown error'}`);
                    }
                } catch (error) {
                    setMessage(`An error occurred: ${(error as Error).message}`);
                } finally {
                    setIsSubmitting(false);
                }
            };

            return (
                <div className="landing-page">
                    <header className="hero-section">
                        <div className="hero-content">
                            <h1>{product_info_content.headline}</h1>
                            <p>{product_info_content.description}</p>
                            <img src={product_info_content.imageUrl} alt={product_info_content.imageAlt} className="product-image" />
                        </div>
                    </header>

                    <section className="features-section">
                        <h2>Key Features</h2>
                        <div className="features-grid">
                            {product_info_content.features.map((feature, index) => (
                                <div key={index} className="feature-item">
                                    <p>{feature}</p>
                                </div>
                            ))}
                        </div>
                    </section>

                    <section className="cta-section">
                        <h2>Stay Updated!</h2>
                        <p>Enter your email to receive exclusive updates and offers.</p>
                        <form onSubmit={handleSubmit} className="email-form">
                            <input
                                type="email"
                                placeholder="Your email address"
                                value={email}
                                onChange={(e: React.ChangeEvent<HTMLInputElement>) => setEmail(e.target.value)}
                                required
                                disabled={isSubmitting}
                            />
                            <button type="submit" disabled={isSubmitting}>
                                {isSubmitting ? 'Submitting...' : 'Subscribe'}
                            </button>
                        </form>
                        {message && <p className="form-message">{message}</p>}
                    </section>

                    <footer className="footer-section">
                        <p>&copy; {new Date().getFullYear()} {product_info_content.title}. All rights reserved.</p>
                    </footer>
                </div>
            );
        };

        // --- Dynamic Input Handling for ReactDOM.render ---
        const rootElement = document.getElementById('root');

        if (rootElement) {
            // Retrieve inputs dynamically from data attributes on the root element
            const productInfoString = rootElement.dataset.productInfo;
            const apiEndpoint = rootElement.dataset.apiEndpoint;

            let productInfo: ProductInfoContent = {
                title: "Default Product",
                headline: "Welcome to Our Product!",
                description: "A simple landing page for your amazing product.",
                imageUrl: "https://via.placeholder.com/800x450/cccccc/000000?text=Placeholder+Image",
                imageAlt: "Placeholder Product Image",
                features: ["Feature 1", "Feature 2", "Feature 3"]
            };

            if (productInfoString) {
                try {
                    productInfo = JSON.parse(productInfoString);
                } catch (e) {
                    console.error("Failed to parse product_info_content from data attribute:", e);
                }
            }

            const backendApiEndpoint = apiEndpoint || 'https://api.example.com/default-lead-capture';

            ReactDOM.render(
                <LandingPage
                    product_info_content={productInfo}
                    backend_api_endpoint={backendApiEndpoint}
                />,
                rootElement
            );
        }
    </script>
</body>
</html>
