#include <string>
#include <iostream>
#include <stdexcept>
#include <curl/curl.h>
#include <nlohmann/json.hpp>
#include <utility> // For std::pair

// Struct to hold API credentials and configuration
// Corresponds to the 'Configuration(Credentials)' abstract type.
struct AIAPICredentials {
    std::string api_key;
    std::string api_endpoint;
    std::string model_name; // e.g., "gpt-4o"
    // Add other relevant configuration if needed, e.g., timeout, proxy
};

// Callback function for CURL to write received data into a std::string
size_t WriteCallback(void* contents, size_t size, size_t nmemb, void* userp) {
    ((std::string*)userp)->append((char*)contents, size * nmemb);
    return size * nmemb;
}

/**
 * @brief Executes an API call to an external AI model.
 *
 * This function handles authentication, request formatting, API call execution,
 * response parsing, and error management using libcurl and nlohmann/json.
 *
 * IMPORTANT: This function assumes that `curl_global_init(CURL_GLOBAL_DEFAULT);`
 * has been called once at the application's startup, and `curl_global_cleanup();`
 * will be called once at the application's shutdown. It does NOT perform global
 * libcurl initialization/cleanup itself to avoid performance and thread-safety issues.
 *
 * @param ai_api_credentials API keys and endpoint details for external AI services.
 * @param ai_prompt_payload The structured prompt and parameters to send to the AI model.
 * @return A std::pair containing:
 *         - nlohmann::json: The raw response from the external AI model if successful.
 *                           Will be an empty JSON object if an error occurred.
 *         - std::string: An error message if the API call fails, empty otherwise.
 */
std::pair<nlohmann::json, std::string> execute_ai_connector_005(
    const AIAPICredentials& ai_api_credentials,
    const nlohmann::json& ai_prompt_payload) {

    nlohmann::json ai_model_response = nlohmann::json::object(); // Initialize as empty object
    std::string api_call_error;

    CURL* curl = nullptr;
    CURLcode res;
    std::string readBuffer;
    struct curl_slist* headers = nullptr;

    // Initialize a CURL easy handle for this specific request
    curl = curl_easy_init();
    if (!curl) {
        api_call_error = "Failed to initialize CURL easy handle.";
        return {ai_model_response, api_call_error};
    }

    try {
        // Set the target URL for the API endpoint
        curl_easy_setopt(curl, CURLOPT_URL, ai_api_credentials.api_endpoint.c_str());

        // Set HTTP headers: Content-Type and Authorization
        headers = curl_slist_append(headers, "Content-Type: application/json");
        std::string auth_header = "Authorization: Bearer " + ai_api_credentials.api_key;
        headers = curl_slist_append(headers, auth_header.c_str());
        curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headers);

        // Set the POST data (the JSON payload)
        std::string payload_str = ai_prompt_payload.dump();
        curl_easy_setopt(curl, CURLOPT_POSTFIELDS, payload_str.c_str());
        curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, payload_str.length());

        // Configure the write callback function to capture the response body
        curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, WriteCallback);
        curl_easy_setopt(curl, CURLOPT_WRITEDATA, &readBuffer);

        // Set a timeout for the entire operation (e.g., 60 seconds)
        curl_easy_setopt(curl, CURLOPT_TIMEOUT, 60L);

        // Perform the actual HTTP request
        res = curl_easy_perform(curl);

        if (res != CURLE_OK) {
            // Handle CURL-specific errors (e.g., network issues, timeouts)
            api_call_error = "CURL request failed: " + std::string(curl_easy_strerror(res));
        } else {
            long http_code = 0;
            curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &http_code); // Get HTTP status code

            if (http_code >= 200 && http_code < 300) {
                // Successful HTTP status code, attempt to parse JSON response
                try {
                    ai_model_response = nlohmann::json::parse(readBuffer);
                } catch (const nlohmann::json::parse_error& e) {
                    api_call_error = "Failed to parse JSON response: " + std::string(e.what()) + ". Raw response: " + readBuffer;
                }
            } else {
                // Unsuccessful HTTP status code
                api_call_error = "API call failed with HTTP status code: " + std::to_string(http_code) + ". Response: " + readBuffer;
            }
        }
    } catch (const std::exception& e) {
        // Catch any other unexpected C++ exceptions
        api_call_error = "An unexpected error occurred during API call: " + std::string(e.what());
    }

    // Cleanup resources associated with this specific CURL easy handle
    if (headers) {
        curl_slist_free_all(headers);
    }
    if (curl) {
        curl_easy_cleanup(curl);
    }

    return {ai_model_response, api_call_error};
}
