#include <iostream>
#include <vector>
#include <string>
#include <algorithm> // For std::remove_if, std::find_if
#include <sstream>   // For std::stringstream to generate IDs
#include <random>    // For random ID generation
#include <chrono>    // For seeding random generator

// --- DataSource Definition ---

/**
 * @brief Enum representing the type of a RAG data source.
 */
enum class DataSourceType {
    WebLink,
    LocalFile
};

/**
 * @brief Converts a DataSourceType enum to its string representation.
 * @param type The DataSourceType to convert.
 * @return A string representing the data source type.
 */
std::string dataSourceTypeToString(DataSourceType type) {
    switch (type) {
        case DataSourceType::WebLink: return "Web Link";
        case DataSourceType::LocalFile: return "Local File";
        default: return "Unknown";
    }
}

/**
 * @brief Structure representing a single RAG data source.
 *        Includes a unique ID, type, and path/URL.
 */
struct DataSource {
    std::string id;
    DataSourceType type;
    std::string path;

    /**
     * @brief Prints the details of the data source to standard output.
     */
    void print() const {
        std::cout << "  ID: " << id
                  << ", Type: " << dataSourceTypeToString(type)
                  << ", Path: " << path << std::endl;
    }

    /**
     * @brief Equality operator for DataSource, primarily compares by ID.
     * @param other The other DataSource to compare with.
     * @return True if IDs are equal, false otherwise.
     */
    bool operator==(const DataSource& other) const {
        return id == other.id;
    }
};

// --- RAGSourceManager Definition ---

/**
 * @brief Manages a collection of RAG data sources.
 *        Provides functionality to add, edit, remove, and retrieve sources.
 */
class RAGSourceManager {
public:
    RAGSourceManager() = default;

    /**
     * @brief Adds a new data source to the manager.
     * @param type The type of the data source (WebLink or LocalFile).
     * @param path The path or URL of the data source.
     */
    void addSource(DataSourceType type, const std::string& path) {
        DataSource newSource;
        newSource.id = generateUniqueId();
        newSource.type = type;
        newSource.path = path;
        sources_.push_back(newSource);
        std::cout << "Source added with ID: " << newSource.id << std::endl;
    }

    /**
     * @brief Edits an existing data source identified by its ID.
     * @param id The unique ID of the source to edit.
     * @param newType The new type for the source.
     * @param newPath The new path/URL for the source.
     * @return True if the source was found and updated, false otherwise.
     */
    bool editSource(const std::string& id, DataSourceType newType, const std::string& newPath) {
        auto it = std::find_if(sources_.begin(), sources_.end(),
                               [&](const DataSource& s) { return s.id == id; });
        if (it != sources_.end()) {
            it->type = newType;
            it->path = newPath;
            std::cout << "Source ID " << id << " updated." << std::endl;
            return true;
        }
        std::cout << "Source with ID " << id << " not found." << std::endl;
        return false;
    }

    /**
     * @brief Removes a data source identified by its ID.
     * @param id The unique ID of the source to remove.
     * @return True if the source was found and removed, false otherwise.
     */
    bool removeSource(const std::string& id) {
        auto initial_size = sources_.size();
        sources_.erase(std::remove_if(sources_.begin(), sources_.end(),
                                      [&](const DataSource& s) { return s.id == id; }),
                       sources_.end());
        if (sources_.size() < initial_size) {
            std::cout << "Source with ID " << id << " removed." << std::endl;
            return true;
        }
        std::cout << "Source with ID " << id << " not found." << std::endl;
        return false;
    }

    /**
     * @brief Retrieves a constant reference to the collection of all configured data sources.
     * @return A const reference to a std::vector of DataSource objects.
     */
    const std::vector<DataSource>& getSources() const {
        return sources_;
    }

    /**
     * @brief Prints all configured data sources to standard output.
     */
    void listSources() const {
        if (sources_.empty()) {
            std::cout << "No RAG data sources configured." << std::endl;
            return;
        }
        std::cout << "--- Configured RAG Data Sources ---" << std::endl;
        for (const auto& source : sources_) {
            source.print();
        }
        std::cout << "-----------------------------------" << std::endl;
    }

private:
    std::vector<DataSource> sources_; ///< The collection of managed data sources.

    /**
     * @brief Generates a unique hexadecimal ID for a new data source.
     *        Uses a random number generator for uniqueness.
     * @return A unique ID string.
     */
    std::string generateUniqueId() {
        static std::random_device rd;
        static std::mt19937_64 generator(rd()); // Use 64-bit generator
        static std::uniform_int_distribution<unsigned long long> distribution(0, -1ULL); // Max unsigned long long

        std::stringstream ss;
        ss << std::hex << distribution(generator);
        return ss.str();
    }
};

// --- UI Helper Functions ---

/**
 * @brief Displays the main menu options to the user.
 */
void displayMenu() {
    std::cout << "\n--- RAG Data Source Management ---" << std::endl;
    std::cout << "1. Add Web Link" << std::endl;
    std::cout << "2. Add Local File" << std::endl;
    std::cout << "3. Edit Source" << std::endl;
    std::cout << "4. Remove Source" << std::endl;
    std::cout << "5. List All Sources" << std::endl;
    std::cout << "6. Exit" << std::endl;
    std::cout << "Enter your choice: ";
}

/**
 * @brief Handles the UI flow for adding a new data source.
 * @param manager The RAGSourceManager instance to interact with.
 * @param type The type of data source to add (WebLink or LocalFile).
 */
void addSourceUI(RAGSourceManager& manager, DataSourceType type) {
    std::string path;
    std::cout << "Enter " << dataSourceTypeToString(type) << " Path/URL: ";
    std::cin.ignore(); // Clear the newline character left by previous std::cin
    std::getline(std::cin, path);
    if (path.empty()) {
        std::cout << "Path/URL cannot be empty. Operation cancelled." << std::endl;
        return;
    }
    manager.addSource(type, path);
}

/**
 * @brief Handles the UI flow for editing an existing data source.
 * @param manager The RAGSourceManager instance to interact with.
 */
void editSourceUI(RAGSourceManager& manager) {
    std::string id;
    std::cout << "Enter ID of source to edit: ";
    std::cin.ignore(); // Clear the newline character
    std::getline(std::cin, id);

    if (id.empty()) {
        std::cout << "ID cannot be empty. Operation cancelled." << std::endl;
        return;
    }

    // Find the source to display current info
    const auto& sources = manager.getSources();
    auto it = std::find_if(sources.begin(), sources.end(),
                           [&](const DataSource& s) { return s.id == id; });

    if (it == sources.end()) {
        std::cout << "Source with ID " << id << " not found." << std::endl;
        return;
    }

    std::cout << "Current Source Info:" << std::endl;
    it->print();

    int typeChoice;
    std::string newPath;

    std::cout << "Enter new type (1 for Web Link, 2 for Local File) [Current: " << dataSourceTypeToString(it->type) << "]: ";
    std::cin >> typeChoice;
    std::cin.ignore(); // Clear newline

    DataSourceType newType = it->type; // Default to current type
    if (typeChoice == 1) {
        newType = DataSourceType::WebLink;
    } else if (typeChoice == 2) {
        newType = DataSourceType::LocalFile;
    } else {
        std::cout << "Invalid type choice. Keeping current type." << std::endl;
    }

    std::cout << "Enter new Path/URL (leave empty to keep current) [Current: " << it->path << "]: ";
    std::getline(std::cin, newPath);
    if (newPath.empty()) {
        newPath = it->path; // Keep old path if new one is empty
    }

    manager.editSource(id, newType, newPath);
}

/**
 * @brief Handles the UI flow for removing a data source.
 * @param manager The RAGSourceManager instance to interact with.
 */
void removeSourceUI(RAGSourceManager& manager) {
    std::string id;
    std::cout << "Enter ID of source to remove: ";
    std::cin.ignore(); // Clear the newline character
    std::getline(std::cin, id);
    if (id.empty()) {
        std::cout << "ID cannot be empty. Operation cancelled." << std::endl;
        return;
    }
    manager.removeSource(id);
}

/**
 * @brief Outputs the configured sources in a structured (JSON-like) format to stdout.
 *        This fulfills the contract's requirement for 'user_configured_rag_sources' output.
 * @param sources The collection of DataSource objects to output.
 */
void outputConfiguredSources(const std::vector<DataSource>& sources) {
    std::cout << "\n--- Final User Configured RAG Sources (JSON-like output) ---" << std::endl;
    std::cout << "[" << std::endl;
    for (size_t i = 0; i < sources.size(); ++i) {
        const auto& s = sources[i];
        std::cout << "  {" << std::endl;
        std::cout << "    \"id\": \"" << s.id << "\"," << std::endl;
        std::cout << "    \"type\": \"" << dataSourceTypeToString(s.type) << "\"," << std::endl;
        std::cout << "    \"path\": \"" << s.path << "\"" << std::endl;
        std::cout << "  }" << (i == sources.size() - 1 ? "" : ",") << std::endl;
    }
    std::cout << "]" << std::endl;
    std::cout << "----------------------------------------------------------" << std::endl;
}

// --- Main Application Loop (UI) ---

/**
 * @brief Main function implementing the console-based user interface for RAG data source management.
 * @return 0 on successful execution.
 */
int main() {
    RAGSourceManager manager;
    int choice;

    do {
        manager.listSources(); // Always show current sources at the start of each loop
        displayMenu();
        
        // Input validation for menu choice
        while (!(std::cin >> choice)) {
            std::cout << "Invalid input. Please enter a number: ";
            std::cin.clear(); // Clear error flags
            std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n'); // Discard invalid input
        }

        switch (choice) {
            case 1: // Add Web Link
                addSourceUI(manager, DataSourceType::WebLink);
                break;
            case 2: // Add Local File
                addSourceUI(manager, DataSourceType::LocalFile);
                break;
            case 3: // Edit Source
                editSourceUI(manager);
                break;
            case 4: // Remove Source
                removeSourceUI(manager);
                break;
            case 5: // List All Sources
                // Already listed at the start of the loop, but can be explicitly called
                manager.listSources();
                break;
            case 6: // Exit
                std::cout << "Exiting RAG Data Source Management. Saving configuration..." << std::endl;
                break;
            default:
                std::cout << "Invalid choice. Please try again." << std::endl;
                break;
        }
        
        if (choice != 6) { // Don't prompt to continue if exiting
            std::cout << "\nPress Enter to continue...";
            std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n'); // Consume any leftover newline
            std::cin.get();    // Wait for user to press Enter
        }
    } while (choice != 6);

    // Output the final configuration as required by the contract
    outputConfiguredSources(manager.getSources());

    return 0;
}
